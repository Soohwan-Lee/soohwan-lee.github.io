<script>
(function () {
  var page = document.querySelector('.pubs-page');
  if (!page) return;

  var LABELS = {
    review:     'Under Review',
    journal:    'Journal Articles & Book Chapters',
    conference: 'Peer-reviewed Conference Papers',
    short:      'Short Papers, Posters & Consortia',
    workshop:   'Workshop Papers',
    other:      'Domestic Venues & Thesis'
  };
  var ORDER = ['review', 'journal', 'conference', 'short', 'workshop', 'other'];

  var items = Array.prototype.slice.call(page.querySelectorAll('.pub-item'));
  var yearNodes = Array.prototype.slice.call(page.children).filter(function (el) {
    return el.tagName === 'H2' || el.classList.contains('pub-item');
  });
  var byType = page.querySelector('.pub-bytype');

  ORDER.forEach(function (cat) {
    var group = items.filter(function (el) { return el.getAttribute('data-cat') === cat; });
    if (!group.length) return;
    var h = document.createElement('h2');
    h.textContent = LABELS[cat];
    byType.appendChild(h);
    group.forEach(function (el) { byType.appendChild(el.cloneNode(true)); });
  });

  var buttons = page.querySelectorAll('.pub-toggle__btn');
  var slider = page.querySelector('.pub-toggle__slider');

  function moveSlider() {
    var active = page.querySelector('.pub-toggle__btn.is-active');
    if (!active || !slider) return;
    slider.style.width = active.offsetWidth + 'px';
    slider.style.transform = 'translateX(' + (active.offsetLeft - slider.offsetLeft) + 'px)';
  }

  function show(view) {
    var byYear = view === 'year';
    yearNodes.forEach(function (el) { el.hidden = !byYear; });
    byType.hidden = byYear;
    Array.prototype.forEach.call(buttons, function (b) {
      var on = b.getAttribute('data-view') === view;
      b.classList.toggle('is-active', on);
      b.setAttribute('aria-selected', on ? 'true' : 'false');
    });
    moveSlider();
    try { localStorage.setItem('pubView', view); } catch (e) {}
  }

  Array.prototype.forEach.call(buttons, function (b) {
    b.addEventListener('click', function () { show(b.getAttribute('data-view')); });
  });

  var saved = 'year';
  try { saved = localStorage.getItem('pubView') || 'year'; } catch (e) {}
  show(saved);
  window.addEventListener('resize', moveSlider);
  if (document.fonts && document.fonts.ready) { document.fonts.ready.then(moveSlider); }
})();
</script>
